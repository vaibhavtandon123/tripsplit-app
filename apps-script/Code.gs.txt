// JSON API router — lets the Cloudflare frontend call api_* functions over HTTP.
function doPost(e){
  var out;
  try{
    var req = JSON.parse((e && e.postData && e.postData.contents) || '{}');
    var fn = req.fn, arg = req.arg;
    if(!/^api_/.test(fn) || typeof globalThis[fn] !== 'function') throw 'Unknown action';
    out = { r: globalThis[fn](arg) };
  }catch(err){
    out = { e: (err && err.message) ? err.message : String(err) };
  }
  return ContentService.createTextOutput(JSON.stringify(out)).setMimeType(ContentService.MimeType.JSON);
}

function ensureSheets_(ss){
  Object.keys(SCHEMA).forEach(function(name){
    var sh=ss.getSheetByName(name);
    if(!sh){ sh=ss.insertSheet(name); sh.getRange(1,1,1,SCHEMA[name].length).setValues([SCHEMA[name]]); sh.setFrozenRows(1); }
    else {
      var lastCol=Math.max(1,sh.getLastColumn());
      var have=sh.getRange(1,1,1,lastCol).getValues()[0].map(String);
      SCHEMA[name].forEach(function(col){
        if(have.indexOf(col)<0){ sh.getRange(1,sh.getLastColumn()+1).setValue(col); have.push(col); }
      });
      sh.setFrozenRows(1);
    }
  });
  var def=ss.getSheetByName('Sheet1'); if(def && def.getLastRow()===0) ss.deleteSheet(def);
  // Keep free-text date columns as PLAIN TEXT so Sheets doesn't auto-convert them to Date objects.
  [['Expenses',['spentOn']],['Trips',['monthLabel','startDate','endDate']]].forEach(function(pair){
    var sh=ss.getSheetByName(pair[0]); if(!sh) return;
    var hdr=sh.getRange(1,1,1,sh.getLastColumn()).getValues()[0], idx={}; hdr.forEach(function(c,i){ idx[String(c)]=i; });
    pair[1].forEach(function(col){ if(idx[col]!=null){ try{ sh.getRange(1,idx[col]+1,sh.getMaxRows(),1).setNumberFormat('@'); }catch(e){} } });
  });
  if(readAll_('Users').length===0) seedAdmin_();
}

function dStr_(v){ if(v instanceof Date) return v.toISOString(); return String(v==null?'':v); }

function headerIdx_(sh){ var cols=sh.getRange(1,1,1,sh.getLastColumn()).getValues()[0], m={}; cols.forEach(function(c,i){ if(c!=='') m[String(c)]=i; }); return m; }

function readAll_(name){
  var sh=sheet_(name), last=sh.getLastRow(); if(last<2) return [];
  var idx=headerIdx_(sh), vals=sh.getRange(2,1,last-1,sh.getLastColumn()).getValues();
  return vals.map(function(r){ var o={}; SCHEMA[name].forEach(function(c){ o[c]=(idx[c]!=null)?r[idx[c]]:''; }); return o; });
}
function appendRow_(name,obj){
  var sh=sheet_(name), idx=headerIdx_(sh), width=sh.getLastColumn(), row=[];
  for(var i=0;i<width;i++) row.push('');
  SCHEMA[name].forEach(function(c){ if(idx[c]!=null && obj[c]!==undefined) row[idx[c]]=obj[c]; });
  sh.appendRow(row);
}
function updateWhere_(name,keyCol,keyVal,patch){
  var sh=sheet_(name), idx=headerIdx_(sh), last=sh.getLastRow(); if(last<2) return false;
  var width=sh.getLastColumn(), rng=sh.getRange(2,1,last-1,width), vals=rng.getValues(), ki=idx[keyCol], done=false;
  for(var r=0;r<vals.length;r++){
    if(String(vals[r][ki])===String(keyVal)){
      Object.keys(patch).forEach(function(c){ if(idx[c]!=null) vals[r][idx[c]]=patch[c]; });
      done=true; break;
    }
  }
  if(done) rng.setValues(vals);
  return done;
}

function api_login(p){
  p=p||{}; var u=(p.username||'').trim().toLowerCase(), uDigits=u.replace(/[^0-9]/g,'');
  var user=readAll_('Users').filter(function(x){
    return String(x.username).toLowerCase()===u
      || String(x.email).toLowerCase()===u
      || (uDigits.length>=10 && String(x.mobile||'').replace(/[^0-9]/g,'').slice(-10)===uDigits.slice(-10));
  })[0];
  if(!user) throw 'No account with that username, email or phone.';
  if(hash_(p.password,user.salt)!==user.passHash) throw 'Wrong password.';
  var token=newToken_(), exp=new Date(Date.now()+TOKEN_DAYS*864e5).toISOString();
  updateWhere_('Users','userId',user.userId,{tokenHash:hash_(token,user.salt),tokenExp:exp,lastLogin:now_()});
  return {token:token,user:publicUser_(user)};
}
function userByToken_(token){
  if(!token) throw 'Please log in again.';
  var users=readAll_('Users');
  for(var i=0;i<users.length;i++){ var x=users[i];
    if(x.tokenHash && hash_(token,x.salt)===x.tokenHash){
      if(x.tokenExp && new Date(x.tokenExp)<new Date()) throw 'Session expired. Please log in again.';
      return x;
    }
  }
  throw 'Please log in again.';
}

function api_resetPassword(p){
  var code=CacheService.getScriptCache().get('reset-'+p.userId);
  if(!code||code!==String(p.code)) throw 'Wrong or expired code.';
  var user=readAll_('Users').filter(function(x){return x.userId===p.userId;})[0];
  if(!user) throw 'Account not found.';
  updateWhere_('Users','userId',user.userId,{passHash:hash_(p.password,user.salt)});
  CacheService.getScriptCache().remove('reset-'+p.userId);
  return {ok:true};
}

// Convert a PDF blob to text via Drive OCR/convert (needs UrlFetch scope).
function pdfToText_(blob){
  try{
    var token=ScriptApp.getOAuthToken();
    var meta={name:'tmp-'+Utilities.getUuid(),mimeType:'application/vnd.google-apps.document'};
    var boundary='b'+Utilities.getUuid();
    var pre=Utilities.newBlob('--'+boundary+'\r\nContent-Type: application/json; charset=UTF-8\r\n\r\n'+JSON.stringify(meta)+'\r\n--'+boundary+'\r\nContent-Type: application/pdf\r\n\r\n').getBytes();
    var post=Utilities.newBlob('\r\n--'+boundary+'--').getBytes();
    var payload=pre.concat(blob.getBytes()).concat(post);
    var res=UrlFetchApp.fetch('https://www.googleapis.com/upload/drive/v3/files?uploadType=multipart&ocrLanguage=en',
      {method:'post',contentType:'multipart/related; boundary='+boundary,payload:payload,headers:{Authorization:'Bearer '+token},muteHttpExceptions:true});
    var id=(JSON.parse(res.getContentText())||{}).id; if(!id) return '';
    var text=''; try{ text=DocumentApp.openById(id).getBody().getText(); }catch(e){}
    try{ DriveApp.getFileById(id).setTrashed(true); }catch(e){}
    return text;
  }catch(e){ return ''; }
}
function parseTicketText_(text){
  var out={mode:'',pnr:'',train:'',trainNo:'',route:'',from:'',to:'',date:'',cls:'',fare:0,passengers:[]};
  if(!text) return out;
  var T=text.replace(/ /g,' ');
  if(/IRCTC|Indian Railway|Reservation Slip|Coach|Berth|SLEEPER|Railway/i.test(T)) out.mode='train';
  else if(/Boarding Pass|Flight|Airlines|Indigo|Air India|Vistara|SpiceJet|Akasa/i.test(T)) out.mode='flight';
  else if(/\bBus\b|redBus|AbhiBus|Boarding Point|Sleeper Bus/i.test(T)) out.mode='bus';
  var pnr=T.match(/PNR[^0-9]{0,40}?(\d{10})/i) || T.match(/(?:^|[^0-9])(\d{10})(?:[^0-9]|$)/);
  if(pnr) out.pnr=pnr[1];
  var trn=T.match(/\b(\d{5})\s*\/\s*([A-Z][A-Z0-9().&' ]+?(?:EXP(?:RESS)?|MAIL|SPL|SPECIAL|SUPERFAST|RAJDHANI|SHATABDI|DURONTO|GARIB\s?RATH|INTERCITY|HUMSAFAR|TEJAS|VANDE\s?BHARAT))\b/);
  if(trn){ out.trainNo=trn[1]; out.train=(trn[1]+' '+trn[2]).replace(/\s+/g,' ').trim(); }
  else { var trn2=T.match(/\b(\d{5})\s*\/\s*([A-Z][A-Z0-9().&' ]{3,40})/); if(trn2){ out.trainNo=trn2[1]; out.train=(trn2[1]+' '+trn2[2].trim()).replace(/\s+/g,' '); } }
  var clsP=T.match(/\((1A|2A|3A|3E|2S|SL|CC|EC|FC)\)/);
  if(clsP) out.cls=clsP[1];
  else { var clsN=T.match(/\b(SLEEPER CLASS|THIRD AC|SECOND AC|FIRST AC|CHAIR CAR|EXECUTIVE)\b/i); if(clsN) out.cls=clsN[1]; }
  var headT=T.split(/Passenger Details|Payment Details|INSTRUCTIONS/i)[0];
  var sts=[], re=/([A-Z][A-Za-z.&' ]*?)\s*\(([A-Z]{2,5})\)/g, m;
  while((m=re.exec(headT))){ var code=m[2]; if(['GN','SL','PNR','ERS','GST','NA','IR','HRS','INR','SAC'].indexOf(code)<0 && !/GENERAL|QUOTA|CLASS/i.test(m[1])) sts.push({name:m[1].trim(),code:code}); }
  var uniq=[]; sts.forEach(function(s){ if(!uniq.length||uniq[uniq.length-1].code!==s.code) uniq.push(s); });
  if(uniq.length>=2){ out.from=uniq[0].name+' ('+uniq[0].code+')'; out.to=uniq[uniq.length-1].name+' ('+uniq[uniq.length-1].code+')'; out.route=out.from+' → '+out.to; }
  else if(uniq.length===1){ out.from=uniq[0].name+' ('+uniq[0].code+')'; out.route=out.from; }
  var dt=T.match(/(?:Departure|Start Date)\*?\s*(?:\d{1,2}:\d{2})?\s*(\d{1,2}-[A-Za-z]{3,}-\d{4})/i) || T.match(/\b(\d{1,2}-[A-Za-z]{3}-\d{4})\b/);
  if(dt) out.date=dt[1];
  var far=T.match(/Total Fare\s*\(all inclusive\)\s*(?:Rs\.?|INR|₹)?\s*([\d,]+\.\d{2})/i)
       || T.match(/(?:Total Fare|Total Amount|Grand Total|Amount Paid)\s*(?:\([^)]*\))?\s*(?:Rs\.?|INR|₹)?\s*([\d,]+\.?\d*)/i)
       || T.match(/Ticket Fare\s*(?:Rs\.?|INR|₹)?\s*([\d,]+\.?\d*)/i);
  if(far) out.fare=Number(far[1].replace(/,/g,''))||0;
  var names=[], nre=/(?:^|\n)\s*\d+\.\s+([A-Z][A-Z](?:[A-Z .]*[A-Z])?)(?=\n|\s{2,}|\s+\d|$)/g, nm;
  while((nm=nre.exec(T))){ var nmv=nm[1].trim(); if(nmv.length>2 && !/ACRONYM|WAITLIST|QUOTA|STATUS|PRESCRIBED/i.test(nmv)) names.push(nmv); }
  var seats=[], sre=/\b(?:CNF|RAC|WL)\s*\/\s*([A-Z]{0,3}\d{1,2})\s*\/\s*(\d{1,3})\s*\/\s*([A-Z][A-Z ]*?)(?:\n|CNF|RAC|WL|$)/g, sm, seen={};
  while((sm=sre.exec(T))){ var key=sm[1]+'-'+sm[2]; if(!seen[key]){ seen[key]=1; seats.push({coach:sm[1],seat:sm[2],type:sm[3].trim()}); } }
  names.forEach(function(nn,i){ var s=seats[i]||{}; out.passengers.push({name:nn,coach:s.coach||'',seat:(s.seat||'')+(s.type?' '+s.type:'')}); });
  return out;
}

// One-time (idempotent) migration: old "Tickets" category → train/flight tickets become "Travel",
// everything else (entry tickets) becomes "Entry tickets". Also fixes the trip's category list.
function migrateTripCats_(tripId){
  var sh=sheet_('Expenses'), idx=headerIdx_(sh), last=sh.getLastRow();
  if(last>=2){
    var width=sh.getLastColumn(), rng=sh.getRange(2,1,last-1,width), vals=rng.getValues();
    var ci=idx['category'], ti=idx['tripId'], tj=idx['ticketJSON'], changed=false;
    for(var r=0;r<vals.length;r++){
      if(String(vals[r][ti])!==String(tripId)) continue;
      if(String(vals[r][ci])==='Tickets'){
        var hasTk=vals[r][tj]&&String(vals[r][tj]).length>2;
        vals[r][ci]=hasTk?'Travel':'Entry tickets'; changed=true;
      }
    }
    if(changed) rng.setValues(vals);
  }
  var trip=getTrip_(tripId), cats=[]; try{cats=JSON.parse(trip.cats||'[]');}catch(x){cats=[];}
  var ncats=[],seen={},mutated=false;
  cats.forEach(function(c){ var nc=(c==='Tickets')?'Entry tickets':c; if(nc!==c)mutated=true; if(!seen[nc]){seen[nc]=1;ncats.push(nc);} });
  if(ncats.indexOf('Entry tickets')<0){ ncats.push('Entry tickets'); mutated=true; }
  if(mutated) updateWhere_('Trips','tripId',tripId,{cats:JSON.stringify(ncats)});
}

function kidWeight_(m){ if(!m.isKid) return 1; if(m.kidShare==='none') return 0; if(m.kidShare==='half') return 0.5; return 1; }

function shareMap_(e,memById){
  var out={},keys=Object.keys(e.split||{}); if(!keys.length) return out;
  if(e.splitType==='equal'){
    // weighted by kid settings; split value may itself be an override weight (1/0.5/0)
    var weights={},sum=0;
    keys.forEach(function(k){
      var w; var ov=Number(e.split[k]);
      if(ov===0||ov===0.5||ov===1) w=ov; else w=kidWeight_(memById[k]||{});
      weights[k]=w; sum+=w;
    });
    if(sum<=0){ keys.forEach(function(k){ out[k]=0; }); return out; }
    keys.forEach(function(k){ out[k]=e.amount*weights[k]/sum; });
  } else if(e.splitType==='percent'){ keys.forEach(function(k){ out[k]=e.amount*Number(e.split[k])/100; }); }
  else { keys.forEach(function(k){ out[k]=Number(e.split[k]); }); }
  return out;
}

function computeBalances_(members,expenses,payments){
  var memById={},headOf={},nameOf={},paid={},owe={},iOwe={},iPaid={},pair={};
  members.forEach(function(m){ memById[m.memberId]=m; headOf[m.memberId]=m.headId?m.headId:m.memberId; nameOf[m.memberId]=m.name; iOwe[m.memberId]=0; iPaid[m.memberId]=0; if(!m.headId){paid[m.memberId]=0;owe[m.memberId]=0;} });
  var byCategory={},byDay={},total=0;
  expenses.filter(function(e){return !e.deleted;}).forEach(function(e){
    total+=e.amount;
    byCategory[e.category]=(byCategory[e.category]||0)+e.amount;
    var day=e.spentOn||String(e.addedAt).slice(0,10); byDay[day]=(byDay[day]||0)+e.amount;
    Object.keys(e.payers||{}).forEach(function(pid){ var h=headOf[pid]||pid; if(paid[h]!=null) paid[h]+=Number(e.payers[pid])||0; if(iPaid[pid]!=null) iPaid[pid]+=Number(e.payers[pid])||0; });
    var sh=shareMap_(e,memById);
    Object.keys(sh).forEach(function(mid){ var h=headOf[mid]||mid; if(owe[h]!=null) owe[h]+=sh[mid]; if(iOwe[mid]!=null) iOwe[mid]+=sh[mid]; });
    // pairwise (direct / non-simplified) debts at head level
    var payByHead={},pTot=0;
    Object.keys(e.payers||{}).forEach(function(pid){ var a=Number(e.payers[pid])||0; var h=headOf[pid]||pid; payByHead[h]=(payByHead[h]||0)+a; pTot+=a; });
    if(pTot>0){ var shByHead={}; Object.keys(sh).forEach(function(mid){ var h=headOf[mid]||mid; shByHead[h]=(shByHead[h]||0)+sh[mid]; });
      Object.keys(shByHead).forEach(function(h){ var s=shByHead[h]; Object.keys(payByHead).forEach(function(k){ if(k!==h){ pair[h]=pair[h]||{}; pair[h][k]=(pair[h][k]||0)+s*(payByHead[k]/pTot); } }); }); }
  });
  var bal={}; Object.keys(paid).forEach(function(h){ bal[h]=round2_(paid[h]-owe[h]); });
  payments.forEach(function(p){ var f=headOf[p.fromId]||p.fromId,t=headOf[p.toId]||p.toId; if(bal[f]!=null)bal[f]=round2_(bal[f]+p.amount); if(bal[t]!=null)bal[t]=round2_(bal[t]-p.amount); });
  var cred=[],debt=[];
  Object.keys(bal).forEach(function(h){ if(bal[h]>0.5)cred.push({id:h,v:bal[h]}); else if(bal[h]<-0.5)debt.push({id:h,v:-bal[h]}); });
  cred.sort(function(a,b){return b.v-a.v;}); debt.sort(function(a,b){return b.v-a.v;});
  var settle=[],i=0,j=0;
  while(i<debt.length&&j<cred.length){ var x=Math.min(debt[i].v,cred[j].v);
    settle.push({fromId:debt[i].id,fromName:nameOf[debt[i].id],toId:cred[j].id,toName:nameOf[cred[j].id],amount:round2_(x)});
    debt[i].v-=x; cred[j].v-=x; if(debt[i].v<0.5)i++; if(cred[j].v<0.5)j++;
  }
  // apply recorded payments to the pairwise matrix, then net each pair → direct settle
  payments.forEach(function(p){ var f=headOf[p.fromId]||p.fromId,t=headOf[p.toId]||p.toId; pair[f]=pair[f]||{}; pair[f][t]=(pair[f][t]||0)-p.amount; });
  var hs=Object.keys(paid),settleDirect=[];
  for(var a2=0;a2<hs.length;a2++){ for(var b2=a2+1;b2<hs.length;b2++){ var A=hs[a2],B=hs[b2];
    var ab=(pair[A]&&pair[A][B])||0, ba=(pair[B]&&pair[B][A])||0, net=round2_(ab-ba);
    if(net>0.5) settleDirect.push({fromId:A,fromName:nameOf[A],toId:B,toName:nameOf[B],amount:net});
    else if(net<-0.5) settleDirect.push({fromId:B,fromName:nameOf[B],toId:A,toName:nameOf[A],amount:round2_(-net)});
  } }
  settleDirect.sort(function(x,y){return y.amount-x.amount;});
  var balances=Object.keys(paid).map(function(h){ return {memberId:h,name:nameOf[h],paid:round2_(paid[h]),owe:round2_(owe[h]),balance:bal[h]}; });
  var memberStats=members.map(function(m){ return {memberId:m.memberId,name:m.name,headId:m.headId||'',isFamily:!!m.isFamily,cost:round2_(iOwe[m.memberId]||0),paid:round2_(iPaid[m.memberId]||0)}; });
  return {balances:balances,settle:settle,settleDirect:settleDirect,memberStats:memberStats,byCategory:byCategory,byDay:byDay,total:round2_(total)};
}
function round2_(n){ return Math.round(n*100)/100; }

  body.appendParagraph('Trip code '+t.code+'   ·   '+d.members.length+' people   ·   Generated '+new Date().toLocaleString('en-IN')).setFontSize(9).setForegroundColor('#6B7078');
  body.appendHorizontalRule();
  body.appendParagraph('Summary').setHeading(DocumentApp.ParagraphHeading.HEADING2);
  styleKV_(body.appendTable([['Total spent',fmtMoney_(d.total,cur)],['People',String(d.members.length)],
    ['Budget',t.budget?fmtMoney_(t.budget,cur):'—'],['Per person (avg)',fmtMoney_(d.total/Math.max(1,countHeads_(d.members)),cur)]]));

  body.appendParagraph('Spending by category').setHeading(DocumentApp.ParagraphHeading.HEADING2);

}

function backup_(force){
  try{
    var props=PropertiesService.getScriptProperties();
    var last=Number(props.getProperty('BACKUP_AT')||0);
    if(!force && Date.now()-last < 120000) return;   // throttle: at most once every 2 min (keeps saves fast)
    props.setProperty('BACKUP_AT',String(Date.now()));
    var id=props.getProperty('BACKUP_DOC');
    var doc; if(id){ try{ doc=DocumentApp.openById(id); }catch(e){ doc=null; } }
    if(!doc){ doc=DocumentApp.create('Trip Expense Tracker — Readable Backup'); var f=DriveApp.getFileById(doc.getId()); childFolder_(subFolder_(SUB_BACKUP),'.').addFile(f); DriveApp.getRootFolder().removeFile(f); props.setProperty('BACKUP_DOC',doc.getId()); }
    var body=doc.getBody(); body.clear();
    body.appendParagraph('Trip Expense Tracker — Backup').setHeading(DocumentApp.ParagraphHeading.TITLE);
    body.appendParagraph('Last updated: '+new Date().toLocaleString('en-IN')).setFontSize(9).setForegroundColor('#6B7078');
    var users=readAll_('Users'), uName={}; users.forEach(function(u){uName[u.userId]=u.fullName;});
    var members=readAll_('Members'), expenses=readAll_('Expenses'), payments=readAll_('Payments');
    readAll_('Trips').filter(function(t){return t.status!=='deleted';}).forEach(function(t){
      body.appendParagraph(t.name+'  ('+t.code+')').setHeading(DocumentApp.ParagraphHeading.HEADING1);
      var mem=members.filter(function(m){return m.tripId===t.tripId;});
      body.appendParagraph('People: '+mem.map(function(m){return m.displayName+(m.isFamily?' (family)':'')+(m.isKid?' (kid)':'');}).join(', ')).setFontSize(10);
      var te=expenses.filter(function(e){return e.tripId===t.tripId;});
      var rows=[['Date','Expense','Category','Amount','Deleted']];
      te.forEach(function(e){ rows.push([String(e.spentOn||e.addedAt).slice(0,10),e.description,e.category,String(e.amount),e.deleted?'yes':'']); });
      if(te.length) styleTable_(body.appendTable(rows));
      var pm=payments.filter(function(x){return x.tripId===t.tripId;});
      if(pm.length){ body.appendParagraph('Payments:').setFontSize(10).setBold(true);
        pm.forEach(function(x){ body.appendParagraph('• '+String(x.recordedAt).slice(0,10)+': '+x.amount).setFontSize(9); }); }
    });
    doc.saveAndClose();
  }catch(e){ /* backup is best-effort */ }
}
